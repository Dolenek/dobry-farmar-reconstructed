cmake_minimum_required(VERSION 3.20)
project(LiveFarmFieldLayout LANGUAGES CXX)
get_filename_component(FARMAR_ROOT "${CMAKE_CURRENT_SOURCE_DIR}/../.." ABSOLUTE)
add_executable(farm-field-live-layout ../live_farm_field_layout.cpp)
target_compile_features(farm-field-live-layout PRIVATE cxx_std_17)
target_compile_options(farm-field-live-layout PRIVATE /utf-8)
foreach(category ui core resources graphics engine profiles settings audio file_manager xml)
    target_include_directories(farm-field-live-layout PRIVATE "${FARMAR_ROOT}/src/game/${category}")
endforeach()
target_include_directories(farm-field-live-layout PRIVATE "${FARMAR_ROOT}/.local/direct3d8-headers" "${FARMAR_ROOT}/src/jngload")
