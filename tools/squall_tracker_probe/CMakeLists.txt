cmake_minimum_required(VERSION 3.20)
project(SquallTrackerProbe LANGUAGES CXX)
if(NOT MSVC OR NOT CMAKE_SIZEOF_VOID_P EQUAL 4)
    message(FATAL_ERROR "Use MSVC Win32 for original DLL interface")
endif()
get_filename_component(RECONSTRUCTION_ROOT "${CMAKE_CURRENT_LIST_DIR}/../.." ABSOLUTE)
add_executable(squall-tracker-probe "${RECONSTRUCTION_ROOT}/tools/squall_tracker_probe.cpp")
target_compile_features(squall-tracker-probe PRIVATE cxx_std_17)
target_compile_definitions(squall-tracker-probe PRIVATE _USRDLL)
target_compile_options(squall-tracker-probe PRIVATE /utf-8)
target_include_directories(squall-tracker-probe PRIVATE "${RECONSTRUCTION_ROOT}/src/squall/runtime")
add_executable(squall-tracker-layout "${RECONSTRUCTION_ROOT}/tools/squall_tracker_layout.cpp")
target_compile_features(squall-tracker-layout PRIVATE cxx_std_17)
target_compile_options(squall-tracker-layout PRIVATE /utf-8)
target_include_directories(squall-tracker-layout PRIVATE "${RECONSTRUCTION_ROOT}/src/squall/runtime" "${RECONSTRUCTION_ROOT}/src/squall/runtime/SoundFile")
add_executable(squall-tracker-ctor-probe "${RECONSTRUCTION_ROOT}/tools/squall_tracker_ctor_probe.cpp")
target_compile_features(squall-tracker-ctor-probe PRIVATE cxx_std_17)
target_compile_definitions(squall-tracker-ctor-probe PRIVATE _USRDLL)
target_compile_options(squall-tracker-ctor-probe PRIVATE /utf-8)
target_include_directories(squall-tracker-ctor-probe PRIVATE "${RECONSTRUCTION_ROOT}/src/squall/runtime")
add_executable(squall-tracker-stage-probe "${RECONSTRUCTION_ROOT}/tools/squall_tracker_stage_probe.cpp")
target_compile_features(squall-tracker-stage-probe PRIVATE cxx_std_17)
target_compile_options(squall-tracker-stage-probe PRIVATE /utf-8)
add_executable(squall-tracker-wrappers-probe "${RECONSTRUCTION_ROOT}/tools/squall_tracker_wrappers_probe.cpp")
target_compile_features(squall-tracker-wrappers-probe PRIVATE cxx_std_17)
target_compile_options(squall-tracker-wrappers-probe PRIVATE /utf-8)
add_executable(squall-ogg-failure-probe "${RECONSTRUCTION_ROOT}/tools/squall_ogg_failure_probe.cpp")
target_compile_features(squall-ogg-failure-probe PRIVATE cxx_std_17)
target_compile_definitions(squall-ogg-failure-probe PRIVATE _USRDLL)
target_compile_options(squall-ogg-failure-probe PRIVATE /utf-8)
target_include_directories(squall-ogg-failure-probe PRIVATE "${RECONSTRUCTION_ROOT}/src/squall/runtime")
