cmake_minimum_required(VERSION 3.20)
project(SquallTrackerEffectsProbe LANGUAGES CXX)
if(NOT MSVC OR NOT CMAKE_SIZEOF_VOID_P EQUAL 4)
    message(FATAL_ERROR "Use MSVC Win32 for original tracker layout")
endif()
get_filename_component(RECONSTRUCTION_ROOT "${CMAKE_CURRENT_LIST_DIR}/../.." ABSOLUTE)
add_executable(squall-tracker-effects-probe "${RECONSTRUCTION_ROOT}/tools/squall_tracker_effects_probe.cpp")
target_compile_features(squall-tracker-effects-probe PRIVATE cxx_std_17)
target_compile_definitions(squall-tracker-effects-probe PRIVATE _USRDLL)
target_compile_options(squall-tracker-effects-probe PRIVATE /utf-8)
target_include_directories(squall-tracker-effects-probe PRIVATE "${RECONSTRUCTION_ROOT}/src/squall/runtime" "${RECONSTRUCTION_ROOT}/src/squall/runtime/SoundFile")
