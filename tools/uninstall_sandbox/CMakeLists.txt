cmake_minimum_required(VERSION 3.20)
project(FarmarUninstallSandbox LANGUAGES CXX)
if(NOT WIN32 OR NOT MSVC OR NOT CMAKE_SIZEOF_VOID_P EQUAL 4)
 message(FATAL_ERROR "Use MSVC Win32 for the native uninstall diagnostic")
endif()
set(CMAKE_CXX_STANDARD 17)
add_executable(containment-probe containment_probe.cpp entry_injection.cpp uninstall_fixtures.cpp ui_driver.cpp bootstrap_broker.cpp)
target_compile_options(containment-probe PRIVATE /utf-8)
target_compile_definitions(containment-probe PRIVATE _WIN32_WINNT=0x0602)
set_target_properties(containment-probe PROPERTIES MSVC_RUNTIME_LIBRARY MultiThreaded)
target_link_libraries(containment-probe PRIVATE advapi32 ole32 user32 gdi32)

add_library(uninstall-application-guard SHARED application_guard.cpp application_guard.def)
target_compile_options(uninstall-application-guard PRIVATE /utf-8)
target_compile_definitions(uninstall-application-guard PRIVATE _WIN32_WINNT=0x0602)
set_target_properties(uninstall-application-guard PROPERTIES MSVC_RUNTIME_LIBRARY MultiThreaded)
target_link_libraries(uninstall-application-guard PRIVATE advapi32 ole32 shell32 user32)

add_executable(uninstall-guard-client guard_client.cpp)
target_compile_options(uninstall-guard-client PRIVATE /utf-8)
set_target_properties(uninstall-guard-client PROPERTIES MSVC_RUNTIME_LIBRARY MultiThreaded)
target_link_libraries(uninstall-guard-client PRIVATE advapi32 ole32 shell32)

add_executable(uninstall-bootstrap-client bootstrap_client.cpp)
target_compile_options(uninstall-bootstrap-client PRIVATE /utf-8)
set_target_properties(uninstall-bootstrap-client PROPERTIES MSVC_RUNTIME_LIBRARY MultiThreaded)
target_link_libraries(uninstall-bootstrap-client PRIVATE advapi32)
