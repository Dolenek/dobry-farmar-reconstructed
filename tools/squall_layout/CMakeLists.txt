cmake_minimum_required(VERSION 3.20)
project(SquallPublicLayout LANGUAGES CXX)
if(NOT MSVC OR NOT CMAKE_SIZEOF_VOID_P EQUAL 4)
    message(FATAL_ERROR "Squall public ABI metadata requires MSVC Win32")
endif()
add_executable(squall-public-layout ../squall_public_layout.cpp)
target_compile_features(squall-public-layout PRIVATE cxx_std_17)
target_compile_options(squall-public-layout PRIVATE /utf-8)
target_include_directories(squall-public-layout PRIVATE ../../src/squall/runtime)
