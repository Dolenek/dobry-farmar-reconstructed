cmake_minimum_required(VERSION 3.20)
project(SquallPointerFaultProbe LANGUAGES CXX)
if(NOT MSVC OR NOT CMAKE_SIZEOF_VOID_P EQUAL 4)
    message(FATAL_ERROR "Use MSVC Win32 for original DLL interface")
endif()
get_filename_component(RECONSTRUCTION_ROOT "${CMAKE_CURRENT_LIST_DIR}/../.." ABSOLUTE)
add_executable(squall-pointer-fault-probe "${RECONSTRUCTION_ROOT}/tools/squall_pointer_fault_probe.cpp")
target_compile_features(squall-pointer-fault-probe PRIVATE cxx_std_17)
target_compile_options(squall-pointer-fault-probe PRIVATE /utf-8)
target_include_directories(squall-pointer-fault-probe PRIVATE "${RECONSTRUCTION_ROOT}/src/squall/runtime" "${RECONSTRUCTION_ROOT}/tools")
