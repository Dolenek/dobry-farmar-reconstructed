cmake_minimum_required(VERSION 3.20)
project(SourceCrashTrace LANGUAGES CXX)
if(NOT MSVC OR NOT CMAKE_SIZEOF_VOID_P EQUAL 4)
    message(FATAL_ERROR "Use MSVC Win32 for source x86 thread contexts")
endif()
add_executable(source-crash-trace trace.cpp)
target_compile_features(source-crash-trace PRIVATE cxx_std_17)
target_compile_options(source-crash-trace PRIVATE /utf-8)
target_link_libraries(source-crash-trace PRIVATE dbghelp)
