cmake_minimum_required(VERSION 3.20)
project(SquallWmaProbe LANGUAGES CXX)
if(NOT MSVC OR NOT CMAKE_SIZEOF_VOID_P EQUAL 4)
    message(FATAL_ERROR "Use MSVC Win32 for original DLL interface")
endif()
get_filename_component(RECONSTRUCTION_ROOT "${CMAKE_CURRENT_LIST_DIR}/../.." ABSOLUTE)
add_executable(squall-wma-probe "${RECONSTRUCTION_ROOT}/tools/squall_wma_probe.cpp")
target_compile_features(squall-wma-probe PRIVATE cxx_std_17)
target_compile_definitions(squall-wma-probe PRIVATE _USRDLL)
target_compile_options(squall-wma-probe PRIVATE /utf-8)
target_include_directories(squall-wma-probe PRIVATE "${RECONSTRUCTION_ROOT}/src/squall/runtime")
target_link_libraries(squall-wma-probe PRIVATE ole32 strmiids)
