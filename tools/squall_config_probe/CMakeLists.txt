cmake_minimum_required(VERSION 3.20)
project(SquallConfigProbe LANGUAGES CXX)
if(NOT MSVC OR NOT CMAKE_SIZEOF_VOID_P EQUAL 4)
    message(FATAL_ERROR "Use MSVC Win32 for original DLL interface")
endif()
get_filename_component(RECONSTRUCTION_ROOT "${CMAKE_CURRENT_LIST_DIR}/../.." ABSOLUTE)
add_executable(squall-config-reference-test "${RECONSTRUCTION_ROOT}/tools/squall_config_reference_test.cpp")
target_compile_features(squall-config-reference-test PRIVATE cxx_std_17)
target_compile_definitions(squall-config-reference-test PRIVATE _USRDLL)
target_compile_options(squall-config-reference-test PRIVATE /utf-8)
target_include_directories(squall-config-reference-test PRIVATE "${RECONSTRUCTION_ROOT}/src/squall/runtime")
