cmake_minimum_required(VERSION 3.20)
project(SquallLoadFpuProbe LANGUAGES CXX)
if(NOT MSVC OR NOT CMAKE_SIZEOF_VOID_P EQUAL 4)
    message(FATAL_ERROR "Use MSVC Win32 for original DLL interface")
endif()
get_filename_component(RECONSTRUCTION_ROOT "${CMAKE_CURRENT_LIST_DIR}/../.." ABSOLUTE)
add_executable(squall-load-fpu-probe "${RECONSTRUCTION_ROOT}/tools/squall_load_fpu_probe.cpp")
target_compile_features(squall-load-fpu-probe PRIVATE cxx_std_17)
target_compile_options(squall-load-fpu-probe PRIVATE /utf-8 /arch:IA32)
