cmake_minimum_required(VERSION 3.20)
project(GameplayLayout LANGUAGES CXX)
if(NOT MSVC OR NOT CMAKE_SIZEOF_VOID_P EQUAL 4)
    message(FATAL_ERROR "Read-only gameplay observer requires MSVC Win32")
endif()
add_executable(gameplay-runtime-layout ../gameplay_runtime_layout.cpp)
target_compile_features(gameplay-runtime-layout PRIVATE cxx_std_17)
target_compile_options(gameplay-runtime-layout PRIVATE /utf-8)
target_include_directories(gameplay-runtime-layout PRIVATE ../../src/game/engine ../../src/game/core)
